def block(start_prefix, end_prefix=None, *, inclusive_end=True):
    """Return the exact text from the unique line starting with start_prefix up to (and
    including, if inclusive_end) the next line starting with end_prefix."""
    idx = [i for i, l in enumerate(lines) if l.startswith(start_prefix)]
    assert len(idx) == 1, (start_prefix, idx)
    i = idx[0]
    if end_prefix is None:
        return lines[i]
    j = next(k for k in range(i + 1, len(lines)) if lines[k].startswith(end_prefix))
    return "\n".join(lines[i : j + 1] if inclusive_end else lines[i:j])
